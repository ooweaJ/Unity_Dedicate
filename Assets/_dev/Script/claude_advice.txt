# 🏗️ 프로젝트 전체 아키텍처 가이드

본 가이드는 프로젝트의 전반적인 구조를 설명하여, 코드 수정 시 아키텍처 정렬을 유지하기 위함입니다.

## 1. 아키텍처 맵 (High-Level)
- **Backend API**: `http://api.jaewoo98.store` (Login, Gacha, Matchmaking)
- **Core Managers (Singletons)**:
  - `BackendManager`: 백엔드 통신 유틸리티 (Static)
  - `GameDataManager`: 마스터 데이터 관리 (ScriptableObject 기반)
  - `PlayerDataManager`: 유저 동적 데이터 (Gold, Inventory, Level)
  - `SceneFlowManager`: 씬 전환 및 데이터 전달 (`LoadRequest`)
- **Networking (Mirror)**:
  - `CustomNetworkManager`: 로비/배틀 서버 통합 관리
  - 배틀 서버 할당: `match/acquire` API를 통해 포트 동적 할당

## 2. 데이터 흐름 원칙 (Data Flow)
- **Static Data**: `GameDataManager.Instance.GetItem(id)`를 통해 접근
- **Dynamic Data**: `PlayerDataManager.Instance.GetInventory()` 등으로 접근
- **UI Logic**: 매니저가 UI를 직접 갱신하는 "Push" 방식 지양, UI가 활성화될 때 데이터를 가져오는 "Pull" 방식 지향 (아래 인벤토리 리팩토링 사례 참고)

## 💡 기존 구조의 핵심 문제 (Inventory 사례)
맞아요, 현재 흐름이 뒤집혀 있어요. 문제를 시각적으로 보면:
// ❌ 현재 (Manager가 모든 UI를 알고 직접 밀어넣는 구조)
InventoryUIManager
  └─▶ inventoryUI_Equipment.Init(allItems)   // 매니저가 직접 순회
  └─▶ inventoryUI_Consumable.Init(allItems)  // 강한 결합
  └─▶ inventoryUI_Transcendence.Init(allItems)

// ✅ 원하는 것 (각 UI가 활성화될 때 스스로 데이터를 요청)
EquipmentInventoryUI.OnEnable()  ─▶ "데이터 줘" 델리게이트 호출
ConsumableInventoryUI.OnEnable() ─▶ "데이터 줘" 델리게이트 호출
InventoryUI에 이미 targetType이 있다는 게 포인트예요. 각 인스턴스가 자기 타입을 알고 있으니, 전체 아이템 데이터를 받아서 스스로 필터링하면 됩니다. 매니저는 그냥 데이터 공급자 역할만 하면 돼요.

수정 방향: Func 델리게이트로 Pull 구조로 전환
InventoryUI.cs
csharppublic class InventoryUI : MonoBehaviour
{
    [Header("Settings")]
    [SerializeField] private ItemType targetType;

    [Header("UI References")]
    [SerializeField] private InventorySlot slotPrefab;
    [SerializeField] private Transform contentParent;

    private List<InventorySlot> _slots = new();
    private int _selectedId = -1;

    // ✅ 외부에서 "데이터 어떻게 가져오는지"만 주입받음
    // 호출 시점은 InventoryUI 스스로 결정
    private Func<IEnumerable<PlayerItemData>> _dataProvider;
    private Action<int> _onItemSelected;

    /// <summary>
    /// 매니저가 한 번만 호출. 델리게이트만 등록하고 렌더링은 하지 않음.
    /// </summary>
    public void Setup(Func<IEnumerable<PlayerItemData>> dataProvider, 
                      Action<int> onItemSelected = null)
    {
        _dataProvider = dataProvider;
        _onItemSelected = onItemSelected;
    }

    // ✅ 탭이 열려서 패널이 활성화될 때 스스로 데이터 요청
    private void OnEnable()
    {
        Refresh();
    }

    // ✅ 서버 데이터 갱신 시 외부에서 호출 가능
    public void Refresh()
    {
        if (_dataProvider == null) return;

        // 전체 아이템 중 자신의 타입만 필터링 (targetType 활용)
        var filteredItems = _dataProvider.Invoke()
            .Where(item =>
            {
                var staticData = GameDataManager.Instance.GetItem(item.itemId);
                return staticData != null && staticData.itemType == targetType;
            })
            .ToList();

        RenderSlots(filteredItems);
    }

    private void RenderSlots(List<PlayerItemData> items)
    {
        // 기존 슬롯 제거
        foreach (Transform child in contentParent)
            Destroy(child.gameObject);
        _slots.Clear();

        foreach (var item in items)
        {
            var staticData = GameDataManager.Instance.GetItem(item.itemId);
            var slot = Instantiate(slotPrefab, contentParent);
            slot.Setup(item.itemId, staticData.icon, item.amoutn, OnSlotClickedInternal);
            _slots.Add(slot);
        }

        if (_slots.Count > 0)
            OnSlotClickedInternal(_slots[0].Id);
    }

    private void OnSlotClickedInternal(int id)
    {
        _selectedId = id;
        RefreshSelection(id);
        _onItemSelected?.Invoke(id);
    }

    public void RefreshSelection(int selectedId)
    {
        foreach (var slot in _slots)
            slot.SetSelect(slot.Id == selectedId);
    }
}

InventoryUIManager.cs
csharppublic class InventoryUIManager : MonoBehaviour
{
    [Header("Sub Managers")]
    [SerializeField] private CharacterListManager listManager;
    [SerializeField] private CharacterModelManager modelManager;
    [SerializeField] private CharacterInfoPanel infoPanel;
    [SerializeField] private SidebarManager sidebarManager;

    // ✅ 여러 InventoryUI 인스턴스를 인스펙터에서 연결
    // 각각 targetType만 다르게 설정된 동일한 컴포넌트
    [Header("Item Inventory UIs")]
    [SerializeField] private List<InventoryUI> itemInventoryUIs;

    [Header("Tab Panels")]
    [SerializeField] private List<TabPanelMapping> panelMappings;

    private List<CharacterUIModel> _characterUIModels = new();
    private Dictionary<int, TabPanelMapping> _panelDict = new();

    // ✅ 아이템 데이터 원본을 캐싱 (델리게이트가 참조할 대상)
    private List<PlayerItemData> _cachedItems = new();

    private void OnEnable()
    {
        foreach (var mapping in panelMappings)
            _panelDict[mapping.tabId] = mapping;

        sidebarManager.OnTabChanged += SwitchSubPanel;
        sidebarManager.Init();

        // ✅ Setup은 여기서 한 번만: 델리게이트 등록
        // 이 시점에 _cachedItems는 비어있어도 됨 — 실제 호출은 OnEnable 때
        foreach (var ui in itemInventoryUIs)
        {
            ui.Setup(
                dataProvider: () => _cachedItems,  // 람다가 항상 최신 캐시 참조
                onItemSelected: OnSelectItem
            );
        }
    }

    private void OnDisable()
    {
        sidebarManager.OnTabChanged -= SwitchSubPanel;
    }

    public void InitInventory(IEnumerable<PlayerCharacterData> charDatas, 
                              IEnumerable<PlayerItemData> itemDatas)
    {
        // ✅ 캐시 업데이트만 — 각 UI의 Refresh는 각자가 처리
        _cachedItems = itemDatas.ToList();

        // 캐릭터 초기화 (기존과 동일)
        _characterUIModels = charDatas.Select(s =>
            new CharacterUIModel(s, GameDataManager.Instance.GetCharacter(s.characterId))
        ).ToList();

        listManager.Init(_characterUIModels, 1, OnSelectCharacter);
        if (_characterUIModels.Count > 0)
            OnSelectCharacter(_characterUIModels[0].StaticData.id);

        // ✅ 현재 열려있는 아이템 UI만 즉시 갱신 (나머지는 탭 열릴 때 OnEnable에서 자동)
        foreach (var ui in itemInventoryUIs)
        {
            if (ui.gameObject.activeInHierarchy)
                ui.Refresh();
        }
    }

    private void SwitchSubPanel(int tabId)
    {
        foreach (var mapping in _panelDict.Values)
            mapping.panelsToShow.ForEach(p => p.SetActive(false));

        if (_panelDict.TryGetValue(tabId, out var target))
            target.panelsToShow.ForEach(p => p.SetActive(true));
        // ✅ 탭 전환 → 패널 활성화 → InventoryUI.OnEnable() 자동 발동
        //    별도 Init 호출 불필요
    }

    private void OnSelectCharacter(int id) { /* 기존과 동일 */ }
    private void OnSelectItem(int itemId)
    {
        // 아이템 상세 패널 갱신 등
    }

    public void Open()  { gameObject.SetActive(true); }
    public void Close() { gameObject.SetActive(false); modelManager.ClearAllModels(); }
}

최종 흐름 비교
// ✅ Setup (최초 1회, OnEnable에서)
InventoryUIManager.OnEnable()
  └─ 각 InventoryUI.Setup(() => _cachedItems)   // 람다만 등록, 렌더링 없음

// ✅ 탭 전환 시
SidebarTab 클릭
  └─ SwitchSubPanel() → EquipmentPanel.SetActive(true)
       └─ EquipmentInventoryUI.OnEnable()        // Unity가 자동 호출
            └─ Refresh() → _dataProvider()       // 람다 실행
                 └─ _cachedItems 필터링 → 슬롯 생성

// ✅ 서버 데이터 갱신 시
PlayerInventory.OnChanged
  └─ InventoryController.HandleRefresh()
       └─ InventoryUIManager.InitInventory()
            └─ _cachedItems 업데이트
                 └─ 활성화된 UI만 Refresh() 호출
핵심 포인트:

Setup()은 "어떻게 가져오는지" 만 등록 (1회)
Refresh()는 "실제로 가져와서 그리기" (필요할 때마다)
탭 전환은 OnEnable()이 자연스럽게 처리 — 매니저가 신경 쓸 필요 없음
_cachedItems를 람다로 참조하기 때문에 캐시가 업데이트되면 다음 Refresh() 때 자동으로 최신 데이터 사용
