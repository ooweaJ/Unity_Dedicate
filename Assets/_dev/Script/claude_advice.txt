전체 구조 설계
현업에서 쓰는 방식은 레이어를 명확히 나누고, 각 레이어가 서로 직접 참조하지 않고 이벤트로만 통신하는 구조입니다.
[Data]      ItemRawData (ScriptableObject)
[Slot]      BaseSlot → InventorySlot / EquipmentSlot
[Controller] DragController / ItemActionPopup / ItemInfoPanel
[Manager]   InventoryUIManager (전체 조율)

1. 데이터 레이어
장비 전용 상속 SO를 하나 두는 게 Unity에서 일반적입니다. MonoBehaviour 상속과 달리 데이터 SO 상속은 문제가 없습니다.

2. 슬롯 레이어
csharp// BaseSlot: 아이콘/수량 표시만 담당
public abstract class BaseSlot : MonoBehaviour
{
    [SerializeField] protected Image imgIcon;
    [SerializeField] protected TextMeshProUGUI txtAmount;
    [SerializeField] protected GameObject emptyIndicator;

    public int ItemId { get; private set; } = -1;
    public ItemRawData StaticData { get; private set; }
    public bool IsEmpty => ItemId == -1;

    public virtual void SetItem(int itemId, ItemRawData data, int amount)
    {
        ItemId = itemId;
        StaticData = data;
        imgIcon.sprite = data.icon;
        imgIcon.gameObject.SetActive(true);
        txtAmount.text = amount > 1 ? amount.ToString() : "";
        emptyIndicator.SetActive(false);
    }

    public virtual void Clear()
    {
        ItemId = -1;
        StaticData = null;
        imgIcon.gameObject.SetActive(false);
        txtAmount.text = "";
        emptyIndicator.SetActive(true);
    }
}
csharp// InventorySlot: 클릭/호버/드래그 소스 역할
public class InventorySlot : BaseSlot,
    IPointerClickHandler, IPointerEnterHandler, IPointerExitHandler,
    IBeginDragHandler, IDragHandler, IEndDragHandler
{
    // Manager가 구독할 이벤트들
    public event Action<int, Vector2> OnHoverEnter;
    public event Action              OnHoverExit;
    public event Action<int, Vector2> OnClicked;    // 클릭 (소비템/재료)
    public event Action<int>          OnDragBegin;  // 드래그 시작 (장비)

    public void OnPointerEnter(PointerEventData e)
    {
        if (!IsEmpty) OnHoverEnter?.Invoke(ItemId, e.position);
    }
    public void OnPointerExit(PointerEventData e) => OnHoverExit?.Invoke();

    public void OnPointerClick(PointerEventData e)
    {
        if (IsEmpty) return;
        // 장비는 클릭 무시 (드래그로만 장착)
        if (StaticData.itemType == ItemType.Equipment) return;
        OnClicked?.Invoke(ItemId, e.position);
    }

    public void OnBeginDrag(PointerEventData e)
    {
        if (IsEmpty || StaticData.itemType != ItemType.Equipment) return;
        OnDragBegin?.Invoke(ItemId);
    }

    public void OnDrag(PointerEventData e)
    {
        // DragController가 처리
        DragController.Instance.OnDrag(e.position);
    }

    public void OnEndDrag(PointerEventData e)
    {
        DragController.Instance.OnEndDrag(e);
    }
}
csharp// EquipmentSlot: 드롭 타겟 + 해제 클릭
public class EquipmentSlot : BaseSlot, IDropHandler, IPointerClickHandler
{
    [SerializeField] public EquipmentSlotType acceptedSlotType;

    public event Action<int, EquipmentSlotType> OnItemDropped;  // 장착 요청
    public event Action<int>                     OnItemClicked;  // 해제 요청

    public void OnDrop(PointerEventData e)
    {
        var dragging = DragController.Instance.DraggingData;
        if (dragging == null) return;

        var equipData = dragging as EquipmentRawData;
        if (equipData == null || equipData.slotType != acceptedSlotType) return;

        OnItemDropped?.Invoke(dragging.id, acceptedSlotType);
    }

    public void OnPointerClick(PointerEventData e)
    {
        if (!IsEmpty) OnItemClicked?.Invoke(ItemId); // 해제
    }
}

3. DragController
캔버스 최상단에 붙는 드래그 아이콘 관리자입니다.
csharppublic class DragController : MonoBehaviour
{
    public static DragController Instance { get; private set; }

    [SerializeField] private Image dragIcon;   // Canvas 최상단 이미지
    [SerializeField] private Canvas rootCanvas;

    public ItemRawData DraggingData { get; private set; }

    private void Awake()
    {
        Instance = this;
        dragIcon.gameObject.SetActive(false);
    }

    public void BeginDrag(ItemRawData data)
    {
        DraggingData = data;
        dragIcon.sprite = data.icon;
        dragIcon.gameObject.SetActive(true);
    }

    public void OnDrag(Vector2 screenPos)
    {
        if (DraggingData == null) return;
        dragIcon.transform.position = screenPos;
    }

    public void OnEndDrag(PointerEventData e)
    {
        dragIcon.gameObject.SetActive(false);
        DraggingData = null;
    }
}

4. ItemActionPopup
csharppublic class ItemActionPopup : MonoBehaviour
{
    [SerializeField] private Button btnPrefab;
    [SerializeField] private Transform container;

    private readonly List<Button> _buttons = new();

    public void Show(ItemRawData data, Vector2 position, Action<ItemActionType> onAction)
    {
        foreach (var b in _buttons) Destroy(b.gameObject);
        _buttons.Clear();

        foreach (var action in data.actions)
        {
            var btn = Instantiate(btnPrefab, container);
            btn.GetComponentInChildren<TextMeshProUGUI>().text = action.label;

            var captured = action.type;
            btn.onClick.AddListener(() =>
            {
                onAction?.Invoke(captured);
                Hide();
            });
            _buttons.Add(btn);
        }

        // 위치 보정 (화면 경계)
        ApplyPosition(position);
        gameObject.SetActive(true);
    }

    public void Hide() => gameObject.SetActive(false);

    private void ApplyPosition(Vector2 pos) { /* ItemInfoPanel과 동일한 경계 체크 */ }
}

5. InventoryUIManager (조율)
csharppublic class InventoryUIManager : MonoBehaviour
{
    [SerializeField] private List<InventorySlot>  inventorySlots;
    [SerializeField] private List<EquipmentSlot>  equipmentSlots;
    [SerializeField] private ItemInfoPanel        itemInfoPanel;
    [SerializeField] private ItemActionPopup      actionPopup;

    // 게임 로직 레이어로 올라가는 이벤트
    public event Action<int>                    OnUseItem;
    public event Action<int>                    OnDiscardItem;
    public event Action<int>                    OnOpenTranscendence;
    public event Action<int, EquipmentSlotType> OnEquipItem;
    public event Action<int>                    OnUnequipItem;

    private void OnEnable()
    {
        foreach (var slot in inventorySlots)
        {
            slot.OnHoverEnter += HandleHoverEnter;
            slot.OnHoverExit  += HandleHoverExit;
            slot.OnClicked    += HandleSlotClicked;
            slot.OnDragBegin  += HandleDragBegin;
        }
        foreach (var slot in equipmentSlots)
        {
            slot.OnItemDropped += HandleEquip;
            slot.OnItemClicked += HandleUnequip;
        }
    }

    private void OnDisable()
    {
        foreach (var slot in inventorySlots)
        {
            slot.OnHoverEnter -= HandleHoverEnter;
            slot.OnHoverExit  -= HandleHoverExit;
            slot.OnClicked    -= HandleSlotClicked;
            slot.OnDragBegin  -= HandleDragBegin;
        }
        foreach (var slot in equipmentSlots)
        {
            slot.OnItemDropped -= HandleEquip;
            slot.OnItemClicked -= HandleUnequip;
        }
    }

    // 호버
    private void HandleHoverEnter(int itemId, Vector2 pos)
    {
        var data = GameDataManager.Instance.GetItem(itemId);
        itemInfoPanel.SetData(data, GetAmount(itemId));
        itemInfoPanel.ShowAt(pos);
    }
    private void HandleHoverExit() => itemInfoPanel.Hide();

    // 클릭 (소비템/재료: 액션팝업, 장비: 무시)
    private void HandleSlotClicked(int itemId, Vector2 pos)
    {
        var data = GameDataManager.Instance.GetItem(itemId);
        actionPopup.Show(data, pos, actionType => HandleAction(itemId, actionType));
    }

    // 액션 분기
    private void HandleAction(int itemId, ItemActionType actionType)
    {
        switch (actionType)
        {
            case ItemActionType.Use:               OnUseItem?.Invoke(itemId);            break;
            case ItemActionType.Discard:           OnDiscardItem?.Invoke(itemId);        break;
            case ItemActionType.OpenTranscendence: OnOpenTranscendence?.Invoke(itemId);  break;
        }
    }

    // 드래그 (장비)
    private void HandleDragBegin(int itemId)
    {
        var data = GameDataManager.Instance.GetItem(itemId);
        DragController.Instance.BeginDrag(data);
        itemInfoPanel.Hide();
        actionPopup.Hide();
    }

    // 장착/해제
    private void HandleEquip(int itemId, EquipmentSlotType slotType)
        => OnEquipItem?.Invoke(itemId, slotType);

    private void HandleUnequip(int itemId)
        => OnUnequipItem?.Invoke(itemId);
}

전체 흐름 요약
소비아이템 클릭 → InventorySlot.OnClicked
    → InventoryUIManager.HandleSlotClicked
    → ItemActionPopup.Show (사용/버리기 버튼 동적 생성)
    → 버튼 클릭 → HandleAction → OnUseItem 이벤트 발생
    → 외부 시스템(PlayerInventory)에서 처리

장비 드래그 → InventorySlot.OnDragBegin
    → DragController.BeginDrag (아이콘 따라다님)
    → EquipmentSlot.OnDrop (slotType 맞는지 검증)
    → OnEquipItem 이벤트 → 외부 처리

초월재료 클릭 → ItemActionPopup → OpenTranscendence
    → OnOpenTranscendence 이벤트 → TranscendenceUI 열기
InventoryUIManager는 "무슨 버튼이 눌렸는지"만 알고, 실제 게임 데이터 처리는 이벤트로 외부(PlayerInventory, EquipmentSystem 등)에 위임하는 구조입니다. 덕분에 인벤토리 UI는 게임 로직과 완전히 분리됩니다.
